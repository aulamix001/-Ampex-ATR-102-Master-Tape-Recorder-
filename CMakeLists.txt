cmake_minimum_required(VERSION 3.22)
project(ATR102TapeStudio VERSION 0.1.2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(FetchContent)
if(DEFINED JUCE_SOURCE_DIR)
  add_subdirectory("${JUCE_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/JUCE")
else()
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 4f43011b96eb0636104cb3e433894cda98243626 GIT_SHALLOW FALSE)
  FetchContent_MakeAvailable(JUCE)
endif()
if(CMAKE_CROSSCOMPILING)
  set(ATR_MANIFEST FALSE)
else()
  set(ATR_MANIFEST TRUE)
endif()
juce_add_plugin(ATR102 VST3_AUTO_MANIFEST ${ATR_MANIFEST} COMPANY_NAME "Tape Studio" BUNDLE_ID "studio.tape.atr102" PLUGIN_MANUFACTURER_CODE TpSt PLUGIN_CODE AtR2 FORMATS VST3 Standalone PRODUCT_NAME "ATR-102 Tape Studio" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Distortion)
target_sources(ATR102 PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_options(ATR102 PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
target_compile_definitions(ATR102 PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_DISPLAY_SPLASH_SCREEN=0)
target_link_libraries(ATR102 PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
option(ATR_BUILD_TESTS "Build processor and real VST3 host tests" ON)
if(ATR_BUILD_TESTS)
  enable_testing()
  juce_add_console_app(ATR102Tests PRODUCT_NAME "ATR102Tests")
  target_sources(ATR102Tests PRIVATE tests/ProcessorTests.cpp)
  target_compile_options(ATR102Tests PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
  target_compile_definitions(ATR102Tests PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_MODAL_LOOPS_PERMITTED=1)
  target_link_libraries(ATR102Tests PRIVATE ATR102 juce::juce_audio_utils)
  add_test(NAME processor COMMAND ATR102Tests)
endif()
